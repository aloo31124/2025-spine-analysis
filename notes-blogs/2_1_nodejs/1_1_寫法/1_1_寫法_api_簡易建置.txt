控制層簡化

newshop-server\controllers\product.api.controller.js

    const {getProductList} = require('../services/product.service');

    /* 取得商品列表 */
    exports.getProductList = async (req, res) => {
        const products = await getProductList();
        res.status(200).json({ result: '200', products });
    };

可將 await 簡化為 一行結束

    /* 取得商品列表 */
    exports.getProductList = async (req, res) => {
        res.status(200).json({result:'200', productList: await getProductList()}); ---> await 不可省略, 不然不會等待
    };

服務層 可大幅度 抽離 控制層 邏輯 ---- > 要先封裝

    newshop-server\services\product.service.js

        const db = require('../firestore');

        /* 取得商品列表 */
        exports.getProductList = async () => {
            const snapshot = await db.collection('product').get();
            const productList = [];
            snapshot.forEach(doc => {
                productList.push({id:doc.id, ...doc.data()});
            });
            console.log('productList:', productList);
            return productList;
        }
