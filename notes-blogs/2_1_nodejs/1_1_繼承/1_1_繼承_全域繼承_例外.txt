newshop-server/src/utils/baseHandler.js
    function BaseHandler() {}

    BaseHandler.prototype.safeExecute = async function (fn, ...args) {
        try {
            return await fn.apply(this, args);
        } catch (error) {
            console.error(`[${this.constructor.name}] 執行錯誤: ${error.message}`);
            console.error(error.stack);
            return null; // 可根據需求調整返回值
        }
    };
    module.exports = BaseHandler;

你可以在 `utils` 資料夾中新增一個 `errorHandler.js`，透過裝飾器（Decorator）模式讓所有的函式自動加上錯誤處理邏輯：

### **1️⃣ 新增共用錯誤處理程式**
**`newshop-server/src/utils/errorHandler.js`**
```javascript
module.exports = function withErrorHandler(fn, functionName) {
    return async function(...args) {
        try {
            return await fn(...args);
        } catch (error) {
            console.error(`❌ [${functionName}] 發生錯誤: ${error.message}`);
            console.error(error.stack);
            return null; // 錯誤發生時回傳 null 或自訂錯誤訊息
        }
    };
};
```

---

### **2️⃣ 在 Controllers 中套用**
**`newshop-server/src/controllers/manager/product.api.controller.js`**
```javascript
const withErrorHandler = require('../../utils/errorHandler');
const { getProductList } = require('../../services/product.service');

exports.getProductList = withErrorHandler(async (req, res) => {
    const productList = await getProductList();
    res.status(200).json({
        result: '200',
        productList,
    });
}, 'getProductList');
```

---

### **3️⃣ 在 Services 中套用**
**`newshop-server/src/services/product.service.js`**
```javascript
const db = require('../firestore');
const withErrorHandler = require('../utils/errorHandler');

const getProductList = async () => {
    const snapshot = await db.collection('Product').get();
    const productList = [];
    snapshot.forEach(doc => {
        productList.push({ id: doc.id, ...doc.data() });
    });
    return productList;
};

exports.getProductList = withErrorHandler(getProductList, 'getProductListService');
```

---

### 💡 **優點與特色**：
1. ✅ **集中管理錯誤處理**：所有異常都由 `withErrorHandler` 來捕獲並記錄。
2. ✅ **保持程式不中斷**：出現錯誤時程式繼續執行，不會直接 Crash。
3. ✅ **簡易追蹤錯誤位置**：可直接從錯誤訊息中得知是哪個函式出錯。
4. ✅ **可重複使用**：任何新功能都可以輕鬆加上錯誤處理。

---

### 📝 **範例錯誤輸出示例：**
```bash

TypeError: Cannot read property 'forEach' of undefined
    at getProductList (/src/services/product.service.js:12:21)

```

這樣就能大幅提升程式碼的 **健壯性（Robustness）** 及 **維護性（Maintainability）**！ 😊 🚀
