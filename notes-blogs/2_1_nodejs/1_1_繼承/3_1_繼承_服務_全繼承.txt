如下如何讓 product.service.js 中 getProductList, addProduct,
可以整個函式 原形鏈繼承 BaseHandler.prototype.baseHandler 偵測例外

newshop-server/src/utils/baseHandler.js
    function BaseHandler() {}
    BaseHandler.prototype.baseHandler = async function (fn, ...args) {
        try {
            return await fn.apply(this, args);
        } catch (error) {
            console.error(`[BaseHandler 基本例外偵測][${this.constructor.name}]: 執行錯誤: ${error.message}`);
            console.error(error.stack);
            return null; // 可根據需求調整返回值
        }
    };
    module.exports = BaseHandler;

newshop-server/src/services/product.service.js
    const BaseHandler = require('../utils/baseHandler');
    function ProductService() {}
    ProductService.prototype = Object.create(BaseHandler.prototype);

    ProductService.prototype.getProductList = async function () {
        const snapshot = await db.collection('Product').get();
        const productList = [];
        snapshot.forEach(doc => {
            productList.push({id:doc.id, ...doc.data()});
        });
        return productList;
    }
    ProductService.prototype.addProduct = async function (product) {
        const docRef = await db.collection('Product').add(product);
        const docSnapshot = await docRef.get();
        console.log("新增商品 : ", {id: docRef.id, ...docSnapshot.data()});
        return ({id: docRef.id, ...docSnapshot.data()})
    }
