            const fetchEditProduct = async  () => {
                try {
                    const prodcutResult = await getProduct(id);
                    const editProduct = prodcutResult.data.product;
                    // 設定 編輯商品資訊
                    const name = editProduct.name;
                    const state = "草稿";
                    const type = editProduct.type;
                    const price = editProduct.price;

                    const imgListResult = await getProductImgList(id);
                    const imgList = imgListResult.data.imageList;
                    // 設定 編輯商品資訊
                    setProductParam({
                        name,
                        state,
                        type,
                        price,
                        imgList
                    });
                } catch (error) {
                    alert("取得商品資訊錯誤");
                }
            }
            fetchEditProduct();
        }, []);
