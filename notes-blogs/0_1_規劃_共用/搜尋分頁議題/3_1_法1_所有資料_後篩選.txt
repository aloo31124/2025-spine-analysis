    /* 搜尋商品 */
    exports.searchProduct = async (searchParam, pagingParam) => {
        const { name, state, createDate, type, priceMin, priceMax } = searchParam;
        const { pageIndex = 1, pageSize = 100, sort = 'name' } = pagingParam; // 分頁參數
        //const pageIndex = 1;const pageSize = 3;const sort = 'name' ; // 分頁參數
        let query = db.collection('Product');

        /*
        query 交錯搜尋條件有錯誤, 暫不使用
        if (name) {
            // 使用 includes 進行模糊搜尋
            // query = query.orderBy('name').startAt(name).endAt(name + '\uf8ff'); // 小g提供無法使用 ="=
            query = query.where('name', '==', new RegExp(name, 'i')); // gemini 方法 可使用
        }
        if (state) {
            query = query.where('state', '==', state);
        }
        if (createDate) {
            query = query.where('createDate', '==', createDate);
        }
        if (type) {
            query = query.where('type', '==', type);
        }
        if (priceMin) {
            query = query.where('price', '>=', Number.parseInt(priceMin));
        }
        if (priceMax) {
            query = query.where('price', '<=', Number.parseInt(priceMax));
        }
        */
        const snapshot = await query.get();
        const products = snapshot.docs.map(doc => ({ id: doc.id, ...doc.data() }));

        // 若無 name 搜尋條件，直接返回 Firestore 查詢結果
        if (!name) return products;

        // 使用 Fuse.js 進行模糊搜尋
        const fuse = new Fuse(products, {
            keys: ['name'],
            threshold: 0.3, // 設定匹配度（0 完全匹配，1 模糊匹配）
            includeScore: false,
        });

        return fuse.search(name)
            .sort((a, b) => a.item[sort] > b.item[sort] ? 1 : -1)
            /*
            .filter(result => 
                result.item.state === state
                && result.item.createDate === createDate
                && result.item.type === type
                && result.item.price >= priceMin
                && result.item.price <= priceMax
            )
            */
            .slice((pageIndex - 1) * pageSize, pageIndex * pageSize)
            .map(result => result.item);
    }

    /* 匯入商品 */
    exports.importProduct = async (productList) => {
        const batch = db.batch();
        productList.forEach(product => {
            const docRef = db.collection('Product').doc();
            batch.set(docRef, product);
        });
        await batch.commit();
        console.log("匯入商品 : ", productList);
        return productList;
    }
