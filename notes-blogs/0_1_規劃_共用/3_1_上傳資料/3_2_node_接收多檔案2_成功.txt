這裡提供最簡單的 `Node.js Express` 方案，將上傳圖片的 **所有邏輯集中在 `product.service.js`**，並直接存入 **`C:/shopStorage/product/`** 目錄。  

---

## **📂 1. 安裝 `multer` 處理 `multipart/form-data`**
```sh
npm install express multer path fs-extra
```
- `multer` 👉 處理上傳
- `path` 👉 處理檔案路徑
- `fs-extra` 👉 確保目錄存在

---

## **📂 2. `src/services/product.service.js`**
    ```javascript
    const multer = require('multer');
    const path = require('path');
    const fs = require('fs-extra');

    const UPLOAD_DIR = 'C:/shopStorage/product/';

    // 確保目錄存在
    fs.ensureDirSync(UPLOAD_DIR);

    // 設定 Multer 存儲
    const storage = multer.diskStorage({
        destination: (req, file, cb) => cb(null, UPLOAD_DIR),
        filename: (req, file, cb) => cb(null, `${Date.now()}${path.extname(file.originalname)}`)
    });

    const upload = multer({ storage });

    exports.uploadMiddleware = upload.array('files', 10); // 最多 10 張圖片

    exports.uploadFile = async (req) => {
        if (!req.files || req.files.length === 0) {
            throw new Error('未找到任何上傳檔案');
        }

        return req.files.map(file => ({
            originalName: file.originalname,
            storedName: file.filename,
            path: file.path
        }));
    };
    ```
---

## **📂 3. `src/controllers/manager/product.api.controller.js`**
    ```javascript
    const { uploadFile } = require('../../services/product.service');

    exports.uploadFile = async (req, res) => {
        try {
            const result = await uploadFile(req);
            res.status(200).json({ result: '200', files: result });
        } catch (error) {
            console.error(`[Upload Error] ${error.message}`);
            res.status(500).json({ result: '500', error: error.message });
        }
    };
    ```
---

## **📂 4. `src/index.js`（API 入口）**
    ```javascript
    const express = require('express');
    const app = express();
    const { uploadMiddleware } = require('./services/product.service');
    const productApiController = require('./controllers/manager/product.api.controller');

    app.use(express.json());
    app.use(express.urlencoded({ extended: true }));

    // API: 上傳商品圖片
    app.post('/api/manager/product/upload', uploadMiddleware, productApiController.uploadFile);

    const PORT = 5000;
    app.listen(PORT, () => console.log(`🚀 Server running on port ${PORT}`));
    ```
---

## **📌 5. React 前端請求**
    ```javascript
    import axios from 'axios';

    const uploadProductImages = async (files) => {
        const formData = new FormData();
        files.forEach(file => formData.append('files', file));

        try {
            const response = await axios.post('/api/manager/product/upload', formData, {
                headers: { 'Content-Type': 'multipart/form-data' }
            });
            console.log('✅ 上傳成功:', response.data);
        } catch (error) {
            console.error('❌ 上傳失敗:', error.response?.data || error.message);
        }
    };
    ```
---

## **📌 總結**
✅ **全部邏輯集中在 `product.service.js`**  
✅ **儲存到 `C:/shopStorage/product/`**  
✅ **使用 `multer` 自動處理 `multipart/form-data`**  
✅ **前端 `axios` 可直接傳送 `files` 陣列**  

🚀 **這是最簡單的 `Express` 圖片上傳方案！**
