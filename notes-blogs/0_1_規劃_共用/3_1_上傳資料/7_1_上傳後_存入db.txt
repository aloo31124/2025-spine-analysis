    // 回傳上傳後的圖片資訊
    exports.handleUpload = (req) => {
        if (!req.files || req.files.length === 0) {
            throw new Error('未找到任何上傳檔案');
        }

        return req.files.map(file => ({
            originalName: file.originalname,
            storedName: file.filename,
            path: file.path
        }));
    };

改寫如下:

    exports.uploadFile = async (req) => {
        // 取得上傳後的圖片資訊
        const uploadedFiles = handleUpload(req);

        // 將檔案名稱存入 Firestore (ProductImg 表)
        const savedFiles = await Promise.all(
            uploadedFiles.map(async (file) => {
                return await this.saveProductImage(file.storedName);
            })
        );

        return { success: true, uploadedFiles: savedFiles };
    };

    // 儲存圖片名稱到 Firestore
    exports.saveProductImage = async (imageName) => {
        const docRef = db.collection('ProductImg').doc();
        await docRef.set({ name: imageName });

        return { id: docRef.id, name: imageName };
    };
