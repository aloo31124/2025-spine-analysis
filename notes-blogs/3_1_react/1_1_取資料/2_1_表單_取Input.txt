如下表單送出時, 取得 input 之值,
使用 useRef, useState 兩種方式, 
請分析與比較,
優缺點與使用時機?

// 使用 useRef
function AuthRegisterInfoPage() {
    const mailInput = useRef('');
    const clickSendRegistInfo = async () => {
        const res1 = await postMail(mailInput.current.value);
    }
    return (
        <div>
            <h1>註冊資訊</h1>
            <input type="text"
                placeholder="輸入驗證信箱"
                ref={mailInput}
            />
            <button onClick={clickSendRegistInfo}>送出</button>
        </div>
    )
}
// 使用 useState
function AuthRegisterInfoPage() {
    const [mail, setMail] = useState('');
    const clickSendRegistInfo = async () => {
        const res1 = await postMail(mail);
    }
    return (
        <div>
            <h1>註冊資訊</h1>
            <input 
                type="text"
                placeholder="輸入驗證信箱"
                value={mail}
                onChange={(e) => setMail(e.target.value)}
            />
            <button onClick={clickSendRegistInfo}>送出</button>
        </div>
    )
}
