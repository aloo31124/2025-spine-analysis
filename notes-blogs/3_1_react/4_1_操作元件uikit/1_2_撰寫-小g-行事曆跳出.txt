这里是 **`SearchBar.jsx`** 主要逻辑：
✅ **点击 `createDate` 输入框时，CalendarPicker.jsx 显示**  
✅ **日历固定在 `input` 下方，并对齐右侧**  
✅ **点击 `CalendarPicker.jsx` 外部，日历隐藏**  
✅ **选中日期后，更新 `createDate` 值，并隐藏日历**  

---

### **📌 `SearchBar.jsx`**
```jsx
import React, { useState, useRef, useEffect } from "react";
import CalendarPicker from "../../tools/CalendarPicker/CalendarPicker";
import style from "./SearchBar.module.css";

function SearchBar({ ... }) {
    const [state, setState] = useState("");
    const [createDate, setCreateDate] = useState("");
    const [showCalendar, setShowCalendar] = useState(false);
    const calendarRef = useRef(null);
    const inputRef = useRef(null);

    // 监听点击外部，隐藏 CalendarPicker
    useEffect(() => {
        const handleClickOutside = (event) => {
            if (
                calendarRef.current &&
                !calendarRef.current.contains(event.target) &&
                inputRef.current !== event.target
            ) {
                setShowCalendar(false);
            }
        };
        document.addEventListener("mousedown", handleClickOutside);
        return () => document.removeEventListener("mousedown", handleClickOutside);
    }, []);

    return (
        <div className={style.SearchBar}>
            <div className={style.SearchBarRow}>
                並且
                <input
                    type="text"
                    placeholder="狀態不限"
                    value={state}
                    onChange={(e) => setState(e.target.value)}
                />
                並且
                <input
                    ref={inputRef}
                    type="text"
                    placeholder="建立日期不限"
                    value={createDate}
                    readOnly
                    onClick={() => setShowCalendar(true)}
                />
                {/* 日历组件 */}
                {showCalendar && (
                    <div ref={calendarRef} className={style.calendarWrapper}>
                        <CalendarPicker 
                            onSelectDate={(date) => {
                                setCreateDate(date);
                                setShowCalendar(false);
                            }} 
                        />
                    </div>
                )}
            </div>
            <button onClick={() => console.log("搜尋")}>搜尋</button>
        </div>
    );
}

export default SearchBar;
```

### **✨ 主要功能**
✔ **点击 `createDate`，日历固定在输入框下方，右对齐**  
✔ **选中日期后，赋值给 `createDate`，并隐藏日历**  
✔ **点击外部，隐藏 `CalendarPicker.jsx`**  
✔ **使用 `useRef` & `useEffect` 监听点击事件**
