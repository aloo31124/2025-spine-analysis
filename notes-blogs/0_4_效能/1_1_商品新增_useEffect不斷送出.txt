如下, 子層元件 CreateEditProduct.jsx - productParam 不斷刷新, 造成 商品資訊被刷空。
而 子層元件 productParam 是因為 父元件 ProductAddPage.jsx,

源頭為 父元件 useEffect 偵測 函式 handleUpload 造成。
    * 後續延伸討論如何處理

newshop-client\src\pages\manager\ProductAddPage.jsx
    /* 商品新增 */
    function ProductAddPage() {
        ...

        // 新增商品上傳檔案, 已有商品id
        const handleUpload = async (formData) => {};

        // 刷新圖片
        useEffect(() => {
                console.log(" fetchEditProduct useEffect ")
                // 取得 商品編輯資訊
                const fetchEditProduct = async () => {
                    try {
                        const imgListResult = await getProductImgList(tempId);
                        const imgList = imgListResult.data.imageList;
                        setProductParam({
                            imgList
                        });
                    } catch (error) {
                        //alert("取得商品圖片失敗");
                    }
                }
                fetchEditProduct();
        },[handleUpload]);   // ---------------------------------- > 此處被重複觸發


        return (
            <div className='pageContianer'>
                <CreateEditProduct 
                    productParam={productParam}
                />
            </div>
        );

    }
    export default ProductAddPage;

newshop-client\src\components\manager\CreateEdit\CreateEditProduct.jsx
    import React, { useEffect, useState, useCallback  } from 'react';
    function CreateEditProduct({productParam}) {
        const [name, setName] = useState(null);

        /* 初始商品, 編輯商品資訊 */
        useEffect(() => {
            console.log("productParam useEffect");
            const fetchEditProduct = async () => {
                setName(name || productParam.name);
                // -------------------------------------------------- > 此處 商品資訊被刷空
            }
            fetchEditProduct();
        }, [productParam]);   // ---------------------------- > 此處不斷刷新, 

        return (
            <div>test</div>
        );
    }
    export default CreateEditProduct;
