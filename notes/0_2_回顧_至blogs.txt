    請精要列出 問題程式碼(註解問題), 與解決程式碼(註解解決原理) : 
        const result = productList.map(async product => {
            // 1. `map` 內使用 `async/await` 會直接返回 `Promise` 陣列，而非解析後的結果
            const imgList = await productImgFireStorage.getProductImgList(product.id);
            return {...product, imgList};
        });

        解決:
        const result = await Promise.all( // 使用 Promise.all 等待所有 Promise 解析
            productList.map(async product => {...})
        );
