bug問題: 如下 react專案中,
點選 搜尋區 ProductSearchBar.jsx , 會彈出視窗 SopingSearchDialog.jsx 篩選條件式窗。
試問為何點選 半透明遮罩, 無法關閉視窗?

提示: js觸發氣泡, 父子元件。

解法: 參考最下層


newshop-client\src\components\shopping\ProductSearchBar\ProductSearchBar.jsx
    function ProductSearchBar() {
        const [showSearchDialog, setShowSearchDialog] = useState(false);

        return (
            <div className={style.ProductSearchBar} onClick={() => setShowSearchDialog(true)}>
                <input type="text" 
                    placeholder='輸入搜尋商品'
                />
                <div className={style.ProductSearchItemList}>
                    <img src={filterIcon} />
                    <button className={style.ProductSearchItem}>
                        服裝
                    </button>
                    <button className={style.ProductSearchItem}>
                        500 ~ 2000
                    </button>
                    <button className={style.ProductSearchItem}>
                        雙11免運
                    </button>
                </div>

                {/** 購買商品篩選條件 */}
                {showSearchDialog && (
                    <SopingSearchDialog 
                        isOpen={showSearchDialog}
                        onClose={() => setShowSearchDialog(true)}
                    />
                )}
            </div>
        );
    }export default ProductSearchBar;



newshop-client\src\components\dialog\SopingSearchDialog\SopingSearchDialog.jsx

    function SopingSearchDialog({ isOpen, onClose }) {
        if (!isOpen) return null; // Dialog 關閉時不渲染

        return (
            <div className={dialogStyle.mask} onClick={onClose}>
                <div className={dialogStyle.dialog}>
                    <h2>篩選條件</h2>
                    ...
                    <button>搜尋</button>
                </div>
            </div>
        );    
    }
    export default SopingSearchDialog;

解法:
    原因為 setShowSearchDialog 在太外層, 到只關閉後, 又再次觸發開啟視窗。


newshop-client\src\components\shopping\ProductSearchBar\ProductSearchBar.jsx
    function ProductSearchBar() {
        const [showSearchDialog, setShowSearchDialog] = useState(false);

        return (
            <div className={style.ProductSearchBar} >
                <input type="text" 
                    placeholder='輸入搜尋商品'
                />
                <div className={style.ProductSearchItemList} 
                    onClick={() => setShowSearchDialog(true)} // 修改 setShowSearchDialog 位置
                >
                    ...
                </div>
