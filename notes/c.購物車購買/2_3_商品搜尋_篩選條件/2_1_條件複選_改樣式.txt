                    <div className={_categoryList.some(c => c.isSelected) ? 'tabItem' : 'tabItemClicked'}

                        <div className={category.isSelected ? 'tabItemClicked' : 'tabItem'}
                            key={category.id}

                        >

                        </div>
                    )}
