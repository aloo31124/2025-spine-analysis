newshop-client\src\components\shopping\ProductListPaging\ProductListPaging.jsx
    function ProductListPaging() {
        const [prductList, setProductList] = useState([]);

        useEffect(() => {
            getProductList().then((res) => {
                setProductList(res.data.result);
            }).catch(err => {
                console.log("取得購物商品列表錯誤: ", err)
            })
        }, [])

        return (
            <div className={style.ProductListPaging}>
                {
                    prductList.map(product => {
                        return <ProductCard 
                                    key={product.id}
                                    imgList={product.imgList}
                                    name={product.name}
                                    price={product.price}
                                    type={product.type}
                                />
                    })
                }
                <PaggingBar />
            </div>
        );
    }
    export default ProductListPaging;
