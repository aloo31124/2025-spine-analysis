newshop-server/src/services/auth.service.js
    /* 產生 jwt token */
    exports.createJwtToken = async (req) => {
        const {account} = req.body;
        const user = await getUserByAccount(account);
        const userAgent = req.headers['user-agent'];
        return createToken({userId: user.id, account, userAgent}, '5m');
    }

    /* 驗證 jwt token */
    exports.verifyJwt = (req) => {
        const authHeader = req.headers.authorization;
        if(!authHeader || !authHeader.startsWith('Bearer ')) {
            return null;
        }
        const token = authHeader.split(' ')[1];
        return verifyToken(token);
    }


newshop-server/src/utils/jwt.js
    const jwt = require('jsonwebtoken');
    const SECRET_KEY = 'SECRET_KEY';

    /* 產生 jwt token */
    exports.createToken = (payload, expiresIn = '5m') => {
        return jwt.sign(payload, SECRET_KEY, {expiresIn});
    }
    
    /* 驗證 jwt token */
    exports.verifyToken = (token) => {
        if(!token) return;
        try {
            return jwt.verify(token, SECRET_KEY);
        } catch (err) {
            console.log("驗證 jwt token err : ", err);
            return null
        }
    }
